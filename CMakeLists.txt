cmake_minimum_required(VERSION 3.20)
project(lithocam VERSION 0.0.1 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(lithocam src/scallop.cpp src/kerf.cpp src/dropcutter.cpp src/pathdrop.cpp src/zmap.cpp src/tools.cpp)
target_include_directories(lithocam PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
if(MSVC)
  target_compile_options(lithocam PRIVATE /W4 /permissive-)
else()
  target_compile_options(lithocam PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(dropcutter_cli tools/dropcutter_cli.cpp)
target_link_libraries(dropcutter_cli PRIVATE lithocam)

option(LITHOCAM_TESTS "Build the property tests" ON)
if(LITHOCAM_TESTS)
  enable_testing()
  foreach(t scallop kerf dropcutter dropcutter_profile dropcutter_disc pathdrop zmap tools)
    add_executable(test_${t} tests/test_${t}.cpp)
    target_link_libraries(test_${t} PRIVATE lithocam)
    add_test(NAME ${t} COMMAND test_${t})
  endforeach()
  # Mutation negative control: dropcutter.cpp built with deliberate defects switchable at run time.
  add_executable(test_dropcutter_mutation tests/test_dropcutter_mutation.cpp src/dropcutter.cpp)
  target_include_directories(test_dropcutter_mutation PRIVATE include)
  target_compile_definitions(test_dropcutter_mutation PRIVATE LITHOCAM_MUTATE)
  add_test(NAME dropcutter_mutation COMMAND test_dropcutter_mutation)
endif()
